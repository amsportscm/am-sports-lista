  <script>
    /* =====================================================
       CONFIGURAÇÃO DOS TAMANHOS PRÉ-PROGRAMADOS
       Você pode adicionar ou remover tamanhos aqui.
    ===================================================== */
    const ADULT_SIZES = ['PP', 'P', 'M', 'G', 'GG', 'XG', 'XXG'];
    const KIDS_SIZES = ['INF 2', 'INF 4', 'INF 6', 'INF 8', 'INF 10', 'INF 12', 'INF 14'];
    const BABY_SIZES = ['BABY PP', 'BABY P', 'BABY M', 'BABY G', 'BABY GG', 'BABY XG', 'BABY XXG'];

    let selectedSizeInput = null;

    /* =====================================================
       CRIA UMA NOVA LINHA NA LISTA PRINCIPAL
    ===================================================== */
    function addRow(data = { nome: '', numero: '', tamanho: '' }) {
      const container = document.getElementById('rowsContainer');

      const row = document.createElement('div');
      row.className = 'uniform-row';

      row.innerHTML = `
        <input class="name-input" type="text" placeholder="Nome" value="${data.nome}">
        <input class="number-input" type="text" inputmode="numeric" placeholder="00" value="${data.numero}">
        <input class="size-input" type="text" placeholder="Tam." value="${data.tamanho}" readonly onclick="openSizeModal(this)">
        <button class="delete-line" title="Apagar linha" onclick="deleteRow(this)">×</button>
      `;

      container.appendChild(row);
    }

    /* =====================================================
       APAGA UMA LINHA
    ===================================================== */
    function deleteRow(button) {
      button.closest('.uniform-row').remove();
    }

    /* =====================================================
       ABRE E FECHA O MODAL DE TAMANHOS
    ===================================================== */
    function openSizeModal(input) {
      selectedSizeInput = input;
      document.getElementById('sizeModal').classList.add('active');
    }

    function closeSizeModal() {
      document.getElementById('sizeModal').classList.remove('active');
    }

    function chooseSize(size) {
      if (selectedSizeInput) {
        selectedSizeInput.value = size;
      }
      closeSizeModal();
    }

    /* =====================================================
       MONTA OS BOTÕES DOS TAMANHOS
    ===================================================== */
    function buildSizeButtons() {
      buildButtons('adultSizes', ADULT_SIZES, 'adult');
      buildButtons('kidsSizes', KIDS_SIZES, 'kids');
      buildButtons('babySizes', BABY_SIZES, 'baby');
    }

    function buildButtons(containerId, sizes, className) {
      const container = document.getElementById(containerId);
      container.innerHTML = '';

      sizes.forEach(size => {
        const button = document.createElement('button');
        button.className = className;
        button.textContent = size;
        button.onclick = () => chooseSize(size);
        container.appendChild(button);
      });
    }

    /* =====================================================
       PEGA OS DADOS PREENCHIDOS NA TABELA
    ===================================================== */
    function getListData() {
      const rows = document.querySelectorAll('.uniform-row');
      const data = [];

      rows.forEach(row => {
        const nome = row.querySelector('.name-input').value.trim().toUpperCase();
        const numero = row.querySelector('.number-input').value.trim();
        const tamanho = row.querySelector('.size-input').value.trim().toUpperCase();

        if (nome || numero || tamanho) {
          data.push({
            Nome: nome,
            Número: numero,
            Tamanho: tamanho
          });
        }
      });

      return data;
    }

    /* =====================================================
       MOSTRA A PÁGINA DE RESUMO POR TAMANHO
    ===================================================== */
    function showSummary() {
      const data = getListData();
      const summary = {};

      data.forEach(item => {
        if (item.Tamanho) {
          summary[item.Tamanho] = (summary[item.Tamanho] || 0) + 1;
        }
      });

      const summaryBody = document.getElementById('summaryBody');
      summaryBody.innerHTML = '';

      const orderedSizes = [...ADULT_SIZES, ...KIDS_SIZES, ...BABY_SIZES];
      let hasSummary = false;

      orderedSizes.forEach(size => {
        if (summary[size]) {
          hasSummary = true;
          summaryBody.innerHTML += `
            <tr>
              <td>${size}</td>
              <td>${summary[size]}</td>
            </tr>
          `;
        }
      });

      if (!hasSummary) {
        summaryBody.innerHTML = `
          <tr>
            <td colspan="2" class="empty-summary">Nenhum tamanho preenchido ainda.</td>
          </tr>
        `;
      }

      document.getElementById('grandTotal').textContent = data.length;
      document.getElementById('mainPage').classList.remove('active');
      document.getElementById('summaryPage').classList.add('active');
    }

    function backToMain() {
      document.getElementById('summaryPage').classList.remove('active');
      document.getElementById('mainPage').classList.add('active');
    }

    /* =====================================================
       BAIXA O ARQUIVO EXCEL DA LISTA PRINCIPAL
    ===================================================== */
    function downloadExcel() {
      const data = getListData();

      if (data.length === 0) {
        alert('Preencha pelo menos uma linha antes de baixar a planilha.');
        return;
      }

      const worksheet = XLSX.utils.json_to_sheet(data);
      const workbook = XLSX.utils.book_new();

      XLSX.utils.book_append_sheet(workbook, worksheet, 'Lista Principal');
      XLSX.writeFile(workbook, 'lista-de-uniformes.xlsx');
    }

    /* =====================================================
       LIMPA TODA A LISTA E CRIA LINHAS NOVAS
    ===================================================== */
    function clearAllRows() {
      const confirmClear = confirm('Tem certeza que deseja limpar toda a lista?');

      if (confirmClear) {
        document.getElementById('rowsContainer').innerHTML = '';
        createInitialRows();
      }
    }

    /* =====================================================
       CRIA LINHAS INICIAIS AO ABRIR O SITE
    ===================================================== */
    function createInitialRows() {
      for (let i = 0; i < 15; i++) {
        addRow();
      }
    }

    /* =====================================================
       INICIA O SITE
    ===================================================== */
    buildSizeButtons();
    createInitialRows();

    /* Fecha o modal ao clicar fora da caixa */
    document.getElementById('sizeModal').addEventListener('click', function(event) {
      if (event.target === this) {
        closeSizeModal();
      }
    });
  </script>
